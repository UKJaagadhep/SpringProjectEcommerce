$secretSecure = Read-Host "Keycloak client secret" -AsSecureString
$secret = [System.Net.NetworkCredential]::new("", $secretSecure).Password

$token = (Invoke-RestMethod -Method Post `
  -Uri "http://localhost:8181/realms/spring-microservices-security-realm/protocol/openid-connect/token" `
  -Body @{
    grant_type = "client_credentials"
    client_id = "spring-cloud-client-id"
    client_secret = $secret
  }).access_token

Set-Clipboard -Value $token
